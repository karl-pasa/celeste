# Decision support thresholds

# -----------------------------------------------------------------------------
# Hashing — bcrypt
# -----------------------------------------------------------------------------
BCRYPT_ROUNDS=12

# -----------------------------------------------------------------------------
# Sessions, cache, queue
# -----------------------------------------------------------------------------
# On Vercel the filesystem is read-only apart from /tmp, so sessions and cache
# must live in the database there. Locally, file is fine and faster.
SESSION_DRIVER=database

# Minutes of inactivity before the session ends.
SESSION_LIFETIME=30

# Discard the session cookie when the browser closes.
SESSION_EXPIRE_ON_CLOSE=true

# Encrypt session payloads at rest — with the database driver they sit in a
# table any database read can query.
SESSION_ENCRYPT=true

# HTTPS-only session cookie. Set false ONLY for local HTTP development.
SESSION_SECURE_COOKIE=true

SESSION_PATH=/
SESSION_SAME_SITE=lax

# -----------------------------------------------------------------------------
# Authentication policy
# -----------------------------------------------------------------------------
# Remember-me issues a cookie valid for years and bypasses SESSION_LIFETIME.
# Leave false for the Registrar's shared counter terminal.
CELESTE_ALLOW_REMEMBER=false

# Force a password change while the account is still on its provisioned
# credential. For students that is their student number, which is printed on
# every document this system issues.
CELESTE_REQUIRE_PASSWORD_CHANGE=true

# Leave false until MAIL_* is configured, or students will be locked out of a
# system that cannot send them a verification link.
CELESTE_REQUIRE_EMAIL_VERIFICATION=false
